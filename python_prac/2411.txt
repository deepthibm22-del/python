
# PYTHON OOP CONCEPTS

# 1. CLASS
# A class is a blueprint or template used to create objects.

class Student:
    name = "Deepthi"
    age = 21

    def display(self):
        print(self.name)
        print(self.age)


# 2. OBJECT
# An object is an instance of a class.

class Student:
    name = "Deepthi"
    age = 21

    def display(self):
        print(self.name)
        print(self.age)

s1 = Student()
s1.display()


# 3. SELF KEYWORD
# self represents the current object of a class.

class Student:
    def display(self):
        print("Welcome to Python")

s1 = Student()
s1.display()


# 4. CONSTRUCTOR (__init__)
# A constructor is automatically called when an object is created.

class Student:
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def display(self):
        print("Name:", self.name)
        print("Age:", self.age)

s1 = Student("Deepthi", 21)
s2 = Student("Kavya", 22)

s1.display()
s2.display()


# 5. INHERITANCE
# One class acquires the properties and methods of another class.

class Parent:
    def display(self):
        print("This is parent class")

class Child(Parent):
    def show(self):
        print("This is child class")

obj = Child()
obj.display()
obj.show()


# 6. POLYMORPHISM
# One method name can have different behaviours.

class Dog:
    def sound(self):
        print("Bark")

class Cat:
    def sound(self):
        print("Meow")

d = Dog()
c = Cat()

d.sound()
c.sound()


# 7. ENCAPSULATION
# Combining data and methods inside a class
# and controlling access to data.

class Student:
    def __init__(self):
        self.__marks = 90

    def display(self):
        print("Marks:", self.__marks)

s1 = Student()
s1.display()


# 8. ABSTRACTION
# Hiding implementation details and showing
# only essential features to the user.

from abc import ABC, abstractmethod

class Vehicle(ABC):
    @abstractmethod
    def start(self):
        pass

class Car(Vehicle):
    def start(self):
        print("Car started")

c = Car()
c.start()


# 9. METHOD OVERRIDING
# Child class provides its own implementation
# of a parent class method.

class Parent:
    def display(self):
        print("Parent method")

class Child(Parent):
    def display(self):
        print("Child method")

obj = Child()
obj.display()


# 10. INSTANCE METHOD
# A method that works with instance data using self.

class Student:
    def display(self):
        print("Instance method")

s1 = Student()
s1.display()


# 11. CLASS METHOD
# A method that receives the class as its first argument (cls).

class Student:
    college = "MLAC"

    @classmethod
    def display(cls):
        print(cls.college)

Student.display()


# 12. STATIC METHOD
# A method that does not require self or cls.

class Calculator:
    @staticmethod
    def add(a, b):
        print(a + b)

Calculator.add(10, 20)
