
# 1. Count Even and Odd Numbers in a List

l = [10, 15, 20, 25, 30]
even = 0
odd = 0

for i in l:
    if i % 2 == 0:
        even += 1
    else:
        odd += 1

print("Even numbers:", even)
print("Odd numbers:", odd)


# 2. Separate Positive, Negative and Zero

l = [10, -5, 0, 20, -15, 8, 0]

positive = []
negative = []
zero = []

for i in l:
    if i > 0:
        positive.append(i)
    elif i < 0:
        negative.append(i)
    else:
        zero.append(i)

print("Positive numbers:", positive)
print("Negative numbers:", negative)
print("Zeros:", zero)


# 3. Remove Duplicate Elements from a List

l = [10, 20, 10, 30, 20, 40, 30]
unique = []

for i in l:
    if i not in unique:
        unique.append(i)

print("List after removing duplicates:", unique)


# 4. Rotate a List to the Right by One Position

l = [10, 20, 30, 40, 50]

last = l[-1]

for i in range(len(l)-1, 0, -1):
    l[i] = l[i-1]

l[0] = last

print("Rotated list:", l)


# 5. Check Whether a Number is Prime or Not

num = int(input("Enter the number: "))
count = 0

for i in range(1, num+1):
    if num % i == 0:
        count += 1

if count == 2:
    print("It is a prime number")
else:
    print("It is not a prime number")
